Characteristics between diabetic and non-diabetic groups

Chi-square test for categorical variables

chi_diabetes = read.csv("Diabete datasets/diabetes_binary_health_indicators_BRFSS2015.csv")

cat_vars <- c(
  "HighBP", "HighChol", "CholCheck","Smoker", "Stroke", "HeartDiseaseorAttack",
  "PhysActivity", "Fruits", "Veggies", "HvyAlcoholConsump",
  "AnyHealthcare", "NoDocbcCost", "DiffWalk",
  "Sex", "Age", "Education", "Income"
)
chi_results <- list()

for (v in cat_vars) {
  tbl <- table(chi_diabetes$Diabetes_binary, chi_diabetes[[v]])
  test <- chisq.test(tbl)
  chi_results[[v]] <- test
}

for (v in names(chi_results)) {
  cat("\n====================\n")
  cat("Variable:", v, "\n")
  cat("Chi-square:", chi_results[[v]]$statistic, "\n")
  cat("df:", chi_results[[v]]$parameter, "\n")
  cat("p-value:", chi_results[[v]]$p.value, "\n")
}

Visualization for continuous variables

library(dplyr)
library(tidyr)
library(ggplot2)

df_cont <- 
  chi_diabetes |>
  pivot_longer(cols = all_of(cont_vars),
               names_to = "variable",
               values_to = "value")

ggplot(df_cont,aes(x = factor(Diabetes_binary,
                    levels = c(0, 1),
                    labels = c("Non-diabetes", "Diabetes")),
                    y = value,
                    fill = factor(Diabetes_binary))) +
  geom_boxplot(outlier.alpha = 0.15) +
  facet_wrap(~ variable, scales = "free_y") +
  labs(
    title = "Continuous variables of Diabetes Status",
    x = "Diabetes Status",
    y = "Count"
  ) +
  theme_minimal() +
  theme(legend.position = "none")

Comments on the plot

  • BMI: Diabetic individuals display noticeably higher BMI levels.
  • GenHlth: Diabetic participants report worse general health compared to the non-diabetic group.
  • MentHlth: Diabetic individuals experience more days of poor mental health.
  • PhysHlth: The diabetic group reports substantially more days of poor physical health.
